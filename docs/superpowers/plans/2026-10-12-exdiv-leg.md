## Review Focus

1. **Time exit on the ex-date.** A leg whose hourly bars have a missing session before `E` must still exit no later than the last bar before `E`'s session; the label walks distinct NY dates, so the exit is the close of the 5th distinct session seen, which with a gap could be `E` itself. Task 4 pins: the exit bar's NY date is strictly before `ex_date`, else the row is counted `exit_on_or_after_ex_date` and excluded.
2. **Control overlapping the next ex-date.** With a 63-session quarter the control span `[D+32, D+36]` is clear, but a semi-annual-then-quarterly payer or a special can put an ex-date inside it; Task 3 pins `control_overlaps_ex_date` on any cash dividend (special included).
3. **Empty acquisition.** Zero actions (wrong window, wrong cohort) must be `status: failed`, not a completed study with zero events; Task 4 pins it.
4. **Pairing key.** P2 must pair a leg with its own control (same `pair_id`), never with another symbol's control on the same session; Task 4 pins that the paired mean difference equals the mean of per-pair differences.
5. **Calendar edge for the first events.** A dividend whose decision session is fewer than `atr_window + 1` sessions into the calendar is `outside_calendar`, never a crash; Task 3 pins it.

```python
import numpy as np
import pandas as pd

from agentic_trader.research.apriori.catalog import BootstrapSpec
from agentic_trader.research.apriori.statistics import mean_test, paired_test, trimmed_mean


SPEC = BootstrapSpec(block_mean=10, draws=400, seed=7)


def _frame(values_by_session: dict[date, list[float]]) -> pd.DataFrame:
    rows = [{"session": s, "r_cost": v} for s, values in values_by_session.items() for v in values]
    return pd.DataFrame(rows)


def _sessions(n: int) -> list[date]:
    return [date(2022, 1, 3) + timedelta(days=k) for k in range(n)]


def test_mean_test_holds_on_a_clear_edge_and_reports_counts():
    rng = np.random.default_rng(1)
    frame = _frame({s: [0.3 + 0.1 * rng.standard_normal() for _ in range(2)] for s in _sessions(300)})
    result = mean_test(frame, SPEC)
    assert result["holds"] and result["n"] == 600 and result["n_sessions"] == 300
    assert result["ci90"][0] > 0 and result["finite_draws"] == 400 and 0 <= result["p_one_sided"] <= 1


def test_mean_test_does_not_hold_around_zero():
    rng = np.random.default_rng(2)
    frame = _frame({s: [0.1 * rng.standard_normal()] for s in _sessions(300)})
    assert not mean_test(frame, SPEC)["holds"]


def test_paired_test_keys_both_sides_by_the_control_sessions():
    sessions = _sessions(200)
    leg = _frame({s: [0.5] for s in sessions})
    control = _frame({s: [0.1] for s in sessions[:150]})  # the last 50 leg sessions have no control
    result = paired_test(leg, control, SPEC)
    assert result["n_leg"] == 150 and result["n_control"] == 150 and result["n_sessions"] == 150
    assert abs(result["mean_diff"] - 0.4) < 1e-9 and result["holds"]
    assert abs(result["control_mean_r_cost"] - 0.1) < 1e-9
```

```python
"""Shared a priori statistics: event-weighted means under a stationary session-block bootstrap.

Moved verbatim from ``pead_study`` (short-suppression precedent, ``research/setups/baserates.py``):
``ci90`` = [5th, 95th] percentiles; a criterion holds when its lower bound is above zero; the
paired test resamples each control session once for both means.
"""

from __future__ import annotations

import math

import numpy as np
import pandas as pd

from agentic_trader.research.apriori.catalog import BootstrapSpec
from agentic_trader.research.setups.baserates import _ci90, _p_one_sided_positive, _weighted
from agentic_trader.research.setups.study import _stationary_index_draws


__all__ = ["mean_test", "paired_test", "trimmed_mean"]


def _sums(frame: pd.DataFrame, sessions: pd.Index) -> tuple[np.ndarray, np.ndarray]:
    grouped = frame.groupby("session")["r_cost"].agg(["sum", "count"]).reindex(sessions, fill_value=0)
    return grouped["sum"].to_numpy(float), grouped["count"].to_numpy(float)


def _draws(n: int, spec: BootstrapSpec) -> np.ndarray:
    return _stationary_index_draws(n, spec.block_mean, spec.draws, spec.seed)


def mean_test(leg: pd.DataFrame, spec: BootstrapSpec) -> dict:
    sessions = pd.Index(sorted(leg["session"].unique()))
    sums, counts = _sums(leg, sessions)
    boot = np.array([_weighted(sums, counts, rows) for rows in _draws(len(sessions), spec)], dtype=float)
    ci90 = _ci90(boot)
    mean_r_cost = _weighted(sums, counts, np.arange(len(sessions))) if len(sessions) else float("nan")
    return {
        "n": int(counts.sum()),
        "n_sessions": len(sessions),
        "mean_r_cost": mean_r_cost,
        "ci90": ci90,
        "finite_draws": int(np.isfinite(boot).sum()),
        "p_one_sided": _p_one_sided_positive(boot),
        "holds": bool(np.isfinite(mean_r_cost) and mean_r_cost > 0.0 and np.isfinite(ci90[0]) and ci90[0] > 0.0),
    }


def paired_test(leg: pd.DataFrame, control: pd.DataFrame, spec: BootstrapSpec) -> dict:
    sessions = pd.Index(sorted(control["session"].unique()))
    leg_sums, leg_n = _sums(leg, sessions)
    ctl_sums, ctl_n = _sums(control, sessions)

    def diff(rows: np.ndarray) -> float:
        return _weighted(leg_sums, leg_n, rows) - _weighted(ctl_sums, ctl_n, rows)

    boot = np.array([diff(rows) for rows in _draws(len(sessions), spec)], dtype=float)
    ci90 = _ci90(boot)
    mean_diff = diff(np.arange(len(sessions))) if len(sessions) else float("nan")
    return {
        "n_leg": int(leg_n.sum()),
        "n_control": int(ctl_n.sum()),
        "n_sessions": len(sessions),
        "control_mean_r_cost": _weighted(ctl_sums, ctl_n, np.arange(len(sessions))) if len(sessions) else float("nan"),
        "mean_diff": mean_diff,
        "ci90": ci90,
        "finite_draws": int(np.isfinite(boot).sum()),
        "p_one_sided": _p_one_sided_positive(boot),
        "holds": bool(np.isfinite(mean_diff) and mean_diff > 0.0 and np.isfinite(ci90[0]) and ci90[0] > 0.0),
    }


def trimmed_mean(values: np.ndarray, fraction: float) -> float:
    ordered = np.sort(values[np.isfinite(values)])
    cut = math.floor(fraction * len(ordered))
    kept = ordered[cut : len(ordered) - cut]
    return float(kept.mean()) if kept.size else float("nan")
```

```python
class ExdivTrade(BaseModel, frozen=True, extra="forbid"):
    decision_time_et: str
    entry_lead_sessions: int = Field(ge=1)
    control_offset_sessions: int = Field(ge=1)
    stop_atr_multiple: float = Field(gt=0)
    atr_window: int = Field(ge=2)
    target_r: float = Field(gt=0)

    @field_validator("decision_time_et")
    @classmethod
    def _clock(cls, value: str) -> str:
        if not re.fullmatch(r"\d{2}:\d{2}", value):
            raise ValueError("decision_time_et must be HH:MM")
        time.fromisoformat(value)
        return value

        return self
```

```python
"""Pre-ex-dividend events and their mid-quarter placebo controls from corporate actions and daily bars.

For a non-special cash dividend of symbol ``s`` with ex-date ``E`` on the observed calendar:

- regularity: the previous non-special ex-date ``P`` of ``s`` lies ``min_gap_days..max_gap_days``
  calendar days earlier (the Hartzmark-Solomon "predicted dividend");
- the leg decides at 10:35 on session ``D = E - entry_lead_sessions``; liquidity (raw bars, the live
  dynamic-universe functions) and ATR (adjusted bars) are measured as of ``D - 1``;
- no other corporate action of ``s`` falls within ``structural_exclusion_days`` of ``[D, E]``;
- the control is the same name decided at ``D_c = D + control_offset_sessions``, same gates as of
  ``D_c - 1``, dropped when any ex-date of ``s`` (specials included) falls within one session of its
  span or a structural action is nearby. It carries the leg's ``session`` as the pairing key.

Every excluded dividend (or control) is counted by reason and ex-date year, never silently dropped.
"""

from __future__ import annotations

import math
from collections import Counter, defaultdict
from collections.abc import Sequence
from datetime import date, timedelta

import pandas as pd

from agentic_trader.data.corporate_actions import CorporateAction, CorporateActionKind, between
from agentic_trader.research.apriori.catalog import ExdivEntry
from agentic_trader.research.apriori.pead_events import (
    SUPPORTED_SYMBOL,
    MarketData,
    atr_through,
    by_session,
    decision_at,
)
from agentic_trader.screeners.dynamic_universe import StaticReference, median_dollar_volume, static_reference


__all__ = ["EXDIV_EVENT_COLUMNS", "build_exdiv_events"]

EXDIV_EVENT_COLUMNS: tuple[str, ...] = (
    "pair_id",
    "role",
    "symbol",
    "session",
    "control_session",
    "decision_at",
    "ex_date",
    "previous_ex_date",
    "gap_days",
    "rate",
    "foreign",
    "atr",
    "median_dollar_volume",
    "reference",
)


def build_exdiv_events(
    actions: Sequence[CorporateAction], market: MarketData, entry: ExdivEntry
) -> tuple[pd.DataFrame, dict]:
    days = market.trading_days
    position = {day: i for i, day in enumerate(days)}
    start, end = entry.window.decisions
    lead, offset = entry.trade.entry_lead_sessions, entry.trade.control_offset_sessions
    atr_window = entry.trade.atr_window
    clock = entry.trade.decision_time()
    pad = timedelta(days=entry.event.structural_exclusion_days)
    static_daily = {s: market.liquidity_daily[s] for s in market.static_symbols if s in market.liquidity_daily}
    references: dict[date, StaticReference] = {}
    keyed: dict[str, pd.DataFrame] = {}
    keyed_raw: dict[str, pd.DataFrame] = {}
    reasons: Counter[str] = Counter()
    by_year: dict[str, Counter[int]] = defaultdict(Counter)
    rows: list[dict] = []
    dividends: dict[str, list[CorporateAction]] = defaultdict(list)
    structural: dict[str, list[CorporateAction]] = defaultdict(list)
    for action in actions:
        (dividends if action.kind is CorporateActionKind.CASH_DIVIDEND else structural)[action.symbol].append(action)

    def skip(reason: str, ex: date) -> None:
        reasons[reason] += 1
        by_year[reason][ex.year] += 1

    def gate(symbol: str, d: int) -> tuple[tuple[float, float, float] | None, str | None]:
        """Liquidity and ATR as of ``days[d - 1]``: ``(atr, median_dollar_volume, reference)`` or a reason."""
        as_of = days[d - 1]
        if as_of not in references:
            references[as_of] = static_reference(static_daily, entry.universe.static_percentile, as_of=as_of)
        reference = references[as_of].value
        if reference is None:
            return None, "no_reference"
        raw = market.liquidity_daily[symbol]
        dollar_volume = median_dollar_volume(raw, as_of)
        if dollar_volume is None:
            return None, "short_history"
        liquidity = keyed_raw.setdefault(symbol, by_session(raw))
        close = float(liquidity.at[as_of, "Close"]) if as_of in liquidity.index else math.nan
        if not math.isfinite(close):
            return None, "no_raw_close"
        if close < entry.universe.min_price:
            return None, "illiquid_price"
        if dollar_volume < reference:
            return None, "illiquid_volume"
        sym = keyed.setdefault(symbol, by_session(market.daily[symbol]))
        atr = atr_through(sym, days, d - 1, atr_window)
        if atr is None:
            return None, "no_atr"
        return (atr, dollar_volume, reference), None

    total = legs = controls = 0
    for symbol in sorted(dividends):
        series = sorted(dividends[symbol], key=lambda a: (a.effective_on, a.id))
        total += len(series)
        ex_dates = {a.effective_on for a in series}
        regular: list[CorporateAction] = []
        for action in series:
            if action.special:
                skip("special_dividend", action.effective_on)
            else:
                regular.append(action)
        for k, action in enumerate(regular):
            ex = action.effective_on
            if not SUPPORTED_SYMBOL.fullmatch(symbol):
                skip("unsupported_symbol", ex)
                continue
            if k == 0:
                skip("no_predecessor", ex)
                continue
            previous = regular[k - 1].effective_on
            gap = (ex - previous).days
            if not entry.event.min_gap_days <= gap <= entry.event.max_gap_days:
                skip("irregular_gap", ex)
                continue
            e = position.get(ex)
            if e is None:
                skip("non_session_ex_date", ex)
                continue
            d = e - lead
            if d - 1 - atr_window < 0:
                skip("outside_calendar", ex)
                continue
            session = days[d]
            if not start <= session <= end:
                skip("outside_window", ex)
                continue
            adjusted = market.daily.get(symbol)
            raw = market.liquidity_daily.get(symbol)
            if adjusted is None or adjusted.empty or raw is None or raw.empty:
                skip("no_daily_bars", ex)
                continue
            if between(structural[symbol], symbol, session - pad, ex + pad):
                skip("structural_action_nearby", ex)
                continue
            measured, reason = gate(symbol, d)
            if measured is None:
                skip(reason or "no_atr", ex)
                continue
            base = {
                "pair_id": action.id,
                "symbol": symbol,
                "session": session,
                "ex_date": ex,
                "previous_ex_date": previous,
                "gap_days": gap,
                "rate": action.rate,
                "foreign": action.foreign,
            }
            rows.append(
                {
                    **base,
                    "role": "leg",
                    "control_session": None,
                    "decision_at": decision_at(session, clock),
                    "atr": measured[0],
                    "median_dollar_volume": measured[1],
                    "reference": measured[2],
                }
            )
            legs += 1
            dc = d + offset
            if dc >= len(days):
                skip("control_outside_calendar", ex)
                continue
            control_session = days[dc]
            last = days[min(dc + lead - 1, len(days) - 1)]
            span_start, span_end = days[dc - 1], days[min(dc + lead, len(days) - 1)]
            if any(span_start <= other <= span_end for other in ex_dates):
                skip("control_overlaps_ex_date", ex)
                continue
            if between(structural[symbol], symbol, control_session - pad, last + pad):
                skip("control_structural_action_nearby", ex)
                continue
            measured, reason = gate(symbol, dc)
            if measured is None:
                skip(f"control_{reason or 'no_atr'}", ex)
                continue
            rows.append(
                {
                    **base,
                    "role": "control",
                    "control_session": control_session,
                    "decision_at": decision_at(control_session, clock),
                    "atr": measured[0],
                    "median_dollar_volume": measured[1],
                    "reference": measured[2],
                }
            )
            controls += 1

    frame = pd.DataFrame(rows, columns=list(EXDIV_EVENT_COLUMNS))
    frame["control_session"] = frame["control_session"].astype(object).where(frame["control_session"].notna(), None)
    counts = {
        "dividends": total,
        "legs": legs,
        "controls": controls,
        "unpaired_legs": legs - controls,
        "reasons": dict(reasons),
        "reasons_by_year": {reason: dict(years) for reason, years in by_year.items()},
        "reference_names": {day.isoformat(): ref.names for day, ref in sorted(references.items())},
    }
    return frame, counts
```

```python
def hourly_day(day: date, prices: list[float]) -> pd.DataFrame:
    index = [datetime.combine(day, time(9 + k), tzinfo=ET_TZ).astimezone(UTC) for k in range(len(prices))]
    p = np.asarray(prices, dtype=float)
    return pd.DataFrame({"Open": p, "High": p, "Low": p, "Close": p, "Volume": 1.0}, index=pd.DatetimeIndex(index))


def hourly(days, price_by_day) -> pd.DataFrame:
    return pd.concat([hourly_day(d, [price_by_day(d)] * 7) for d in days])


def _event(symbol, session, *, role="leg", pair_id="p1", control_session=None, atr=1.0, ex_date=None, rate=0.5):

    return {
        "pair_id": pair_id,
        "role": role,
        "symbol": symbol,
        "session": session,
        "control_session": control_session,
        "decision_at": decision_at(decision_day, time(10, 35)),
```

```python
"""Pre-ex-dividend labelling, the frozen P1-P4 pass rule and the study executor (entry ``exdiv``).

One direction (LONG), two roles: every accepted dividend has a leg row and, when its mid-quarter
placebo survived its own gates, a control row with the same ``pair_id`` and ``session``. The
bracket is PEAD's trade block walked by ``label_bracket`` with ``max_hold_sessions`` equal to the
lead, so the time exit is the close of the last cum-dividend session; a label whose exit bar falls
on or after the ex-date (a missing session in the hourly bars) is excluded and counted. The
secondary hold crosses the ex-date on adjusted bars and is descriptive only.

``execute_exdiv_study`` writes ``protocol.json`` and ``manifest.json`` before calling ``build``
(the only provider access) and records any failure as ``status: failed``; an acquisition with no
actions fails the run.
"""

_LABEL_SPAN = timedelta(days=60)
_ROLES = ("leg", "control")


@dataclass(frozen=True)
class ExdivInputs:
    actions: tuple[CorporateAction, ...]
    actions_manifest: Mapping[str, Any]
    cohort: Mapping[str, Any]  # path, sha256, symbols
    market: MarketData
    hourly: Mapping[str, pd.DataFrame]
    bar_failures: Mapping[str, str]
    static_requested: tuple[str, ...]


def _cost_column(cost: float) -> str:
    return f"r_cost_{cost:g}bps"


def _label_columns(entry: ExdivEntry) -> list[str]:
    return [
        *EXDIV_EVENT_COLUMNS,
        "decision_price",
        "hit",
        "r",
        "r_cost",
        *(_cost_column(cost) for cost in entry.costs_bps_per_side),
        "hold_secondary_r_cost",
        "holding_sessions",
        "exit_at",
    ]


def label_exdiv_events(
    events: pd.DataFrame, hourly: Mapping[str, pd.DataFrame], entry: ExdivEntry
) -> tuple[pd.DataFrame, dict]:
    counts: Counter[str] = Counter()
    by_year: dict[str, Counter[int]] = defaultdict(Counter)
    by_role: dict[str, Counter[str]] = defaultdict(Counter)
    rows: list[dict] = []

    def miss(reason: str, event: dict) -> None:
        counts[reason] += 1
        by_year[reason][event["session"].year] += 1
        by_role[reason][event["role"]] += 1

    for event in events.to_dict("records"):
        bars = hourly.get(event["symbol"])
        if bars is None or bars.empty:
            miss("no_hourly_bars", event)
            continue
        when = event["decision_at"]
        index = utc_index(bars)
        window = bars.loc[(index >= when - timedelta(days=1)) & (index <= when + _LABEL_SPAN)]
        price = decision_price(window, when)
        if price is None:
            miss("no_decision_price", event)
            continue
        levels = levels_for("LONG", price, float(event["atr"]), entry)
        if levels is None:
            miss("degenerate_levels", event)
            continue
        costs = {
            _cost_column(cost): label_bracket(
                levels, when, window, max_hold_sessions=entry.trade.max_hold_sessions, cost_bps_per_side=cost
            )
            for cost in entry.costs_bps_per_side
        }
        decisive = costs[_cost_column(entry.decision_cost_bps)]
        if decisive.hit == BracketHit.IMMATURE:
            miss("immature", event)
            continue
        exit_at = decisive.exit_time
        if event["role"] == "leg" and exit_at is not None and exit_at.astimezone(ET_TZ).date() >= event["ex_date"]:
            miss("exit_on_or_after_ex_date", event)
            continue
        secondary = math.nan
        if event["role"] == "leg":
            outcome = label_bracket(
                levels,
                when,
                window,
                max_hold_sessions=entry.trade.secondary_hold_sessions,
                cost_bps_per_side=entry.decision_cost_bps,
            )
            secondary = outcome.r_cost if outcome.r_cost is not None else math.nan
        rows.append(
            {
                **event,
                "decision_price": price,
                "hit": str(decisive.hit),
                "r": decisive.r,
                "r_cost": decisive.r_cost,
                **{column: outcome.r_cost for column, outcome in costs.items()},
                "hold_secondary_r_cost": secondary,
                "holding_sessions": decisive.holding_sessions,
                "exit_at": exit_at,
            }
        )
    summary: dict = dict(counts)
    summary["by_year"] = {reason: dict(years) for reason, years in by_year.items()}
    summary["by_role"] = {reason: dict(roles) for reason, roles in by_role.items()}
    return pd.DataFrame(rows, columns=_label_columns(entry)), summary


def evaluate_exdiv(labels: pd.DataFrame, entry: ExdivEntry) -> dict:
    leg = labels[labels["role"] == "leg"]
    control = labels[labels["role"] == "control"]
    paired = set(leg["pair_id"]) & set(control["pair_id"])
    leg_paired = leg[leg["pair_id"].isin(paired)]
    control_paired = control[control["pair_id"].isin(paired)]
    recent = leg[leg["session"] >= entry.window.recent_from]
    trimmed = trimmed_mean(leg["r_cost"].to_numpy(float), entry.pass_rule.trim_fraction)
    recent_mean = float(recent["r_cost"].mean()) if len(recent) else float("nan")
    p1 = mean_test(leg, entry.bootstrap) if len(leg) else {"n": 0, "holds": False}
    p2 = paired_test(leg_paired, control_paired, entry.bootstrap) if len(leg_paired) else {"n_leg": 0, "holds": False}
    p3 = {
        "trimmed_mean": trimmed,
        "recent_mean": recent_mean,
        "holds": bool(np.isfinite(trimmed) and trimmed > 0 and np.isfinite(recent_mean) and recent_mean > 0),
    }
    p4 = {
        "n": len(leg),
        "n_recent": len(recent),
        "holds": len(leg) >= entry.pass_rule.min_events and len(recent) >= entry.pass_rule.min_recent_events,
    }
    return {
        "p1": p1,
        "p2": p2,
        "p3": p3,
        "p4": p4,
        "passes": all(p["holds"] for p in (p1, p2, p3, p4)),
        "unpaired_legs": len(leg) - len(leg_paired),
    }


def _mean_n(frame: pd.DataFrame, column: str = "r_cost") -> dict:
    values = frame[column].dropna()
    return {"n": len(values), "mean_r_cost": float(values.mean()) if len(values) else float("nan")}


def _terciles(frame: pd.DataFrame, values: pd.Series) -> dict:
    if len(frame) < 3:
        return {}
    buckets = pd.qcut(values.rank(method="first"), 3, labels=["low", "mid", "high"])
    return {str(name): _mean_n(group) for name, group in frame.groupby(buckets, observed=True)}


def _diagnostics(labels: pd.DataFrame) -> dict:
    leg = labels[labels["role"] == "leg"]
    control = labels[labels["role"] == "control"]
    years = pd.to_datetime(leg["session"]).dt.year if len(leg) else pd.Series(dtype=int)
    control_years = pd.to_datetime(control["session"]).dt.year if len(control) else pd.Series(dtype=int)
    per_session = leg.groupby("session").size() if len(leg) else pd.Series(dtype=int)
    return {
        "costs": {column: _mean_n(leg, column) for column in labels.columns if column.startswith("r_cost_")},
        "secondary_hold": _mean_n(leg, "hold_secondary_r_cost"),
        "control": _mean_n(control),
        "by_year": {int(year): _mean_n(group) for year, group in leg.groupby(years)},
        "control_by_year": {int(year): _mean_n(group) for year, group in control.groupby(control_years)},
        "liquidity_terciles": _terciles(leg, leg["median_dollar_volume"]),
        "yield_terciles": _terciles(leg, leg["rate"].astype(float) / leg["decision_price"].astype(float)),
        "by_foreign": {str(flag): _mean_n(group) for flag, group in leg.groupby(leg["foreign"].astype(bool))},
        "hits": {str(k): int(v) for k, v in leg["hit"].value_counts().items()},
        "mean_holding_sessions": float(leg["holding_sessions"].mean()) if len(leg) else float("nan"),
        "events_per_session": {
            "max": int(per_session.max()) if len(per_session) else 0,
            "median": float(per_session.median()) if len(per_session) else 0.0,
            "p90": float(per_session.quantile(0.9)) if len(per_session) else 0.0,
        },
    }


async def execute_exdiv_study(
    loaded: LoadedExdivEntry,
    directory,
    *,
    build: Callable[[], Awaitable[ExdivInputs]],
    environment: dict,
) -> dict:
    entry = loaded.entry
    directory.mkdir(mode=0o700, parents=True, exist_ok=False)
    save_json_report({**entry.model_dump(mode="json"), "sha256": loaded.sha256}, directory / "protocol.json")
    save_json_report(
        {
            "entry_id": entry.id,
            "version": entry.version,
            "sha256": loaded.sha256,
            "environment": environment,
            "started_at": datetime.now(UTC).isoformat(),
            "authorizes_promotion": False,
        },
        directory / "manifest.json",
    )
    try:
        inputs = await build()
        if not inputs.actions:
            raise ValueError("the acquisition returned no corporate actions for the cohort and window")
        events, event_counts = await asyncio.to_thread(build_exdiv_events, inputs.actions, inputs.market, entry)
        reference_names = event_counts.pop("reference_names")
        labels, label_counts = await asyncio.to_thread(label_exdiv_events, events, inputs.hourly, entry)
        label_counts["rows"] = len(labels)
        await asyncio.to_thread(_save_frame, events, directory / "events.csv.gz")
        await asyncio.to_thread(_save_frame, labels, directory / "labels.csv.gz")
        leg = evaluate_exdiv(labels, entry)
        result = {
            "status": "completed",
            "leg": leg,
            "decision": "eligible_for_probe" if leg["passes"] else "failed",
            "diagnostics": _diagnostics(labels),
            "actions": dict(inputs.actions_manifest),
            "cohort": dict(inputs.cohort),
            "universe": universe_summary(inputs.market, inputs.bar_failures, inputs.static_requested, reference_names),
            "events": event_counts,
            "labels": label_counts,
            "bar_failures": dict(inputs.bar_failures),
            "authorizes_promotion": False,
        }
    except Exception as exc:
        result = {"status": "failed", "error": f"{type(exc).__name__}: {exc}", "authorizes_promotion": False}
```

```python
"""Provider access for the exdiv study: trading calendar, cached corporate actions and SIP bars.

Called only from inside ``execute_exdiv_study``'s ``build`` callable, after the manifest exists.
The cohort file is read (and digested) first; part 1's ``fetch_corporate_actions_cached`` supplies
the actions over the entry's process-date window; adjusted daily bars for the cohort (ATR), raw daily
bars for the cohort and static universe (the liquidity gate) and regular-session hourly bars for
symbols with at least one event all reuse the setup study's immutable ``.npz`` cache. A symbol whose
bars fail is recorded in ``bar_failures``, never silently dropped.
"""

from __future__ import annotations

import asyncio
import hashlib
from collections.abc import Awaitable, Callable, Sequence
from datetime import UTC, datetime, time, timedelta
from pathlib import Path

from agentic_trader.data.corporate_actions import CorporateActionsSource
from agentic_trader.research.apriori.catalog import ExdivEntry
from agentic_trader.research.apriori.exdiv_events import build_exdiv_events
from agentic_trader.research.apriori.exdiv_study import ExdivInputs
from agentic_trader.research.apriori.pead_events import SUPPORTED_SYMBOL, MarketData
from agentic_trader.research.apriori.pead_runner import _bars_or_failure, _record
from agentic_trader.research.corporate_actions import cohort_symbols, fetch_corporate_actions_cached
from agentic_trader.research.setups.runner import BarSource, CalendarSource, _claim_cache_range


__all__ = ["build_exdiv_inputs"]

_CALENDAR_PAD = timedelta(days=60)
_HOURLY_LEAD = timedelta(days=7)
_HOURLY_TAIL = timedelta(days=60)


async def build_exdiv_inputs(
    entry: ExdivEntry,
    *,
    bars: BarSource,
    calendar: CalendarSource,
    actions_source: CorporateActionsSource,
    cache_dir: Path,
    static_symbols: Sequence[str],
    pace: Callable[[], Awaitable[None]],
    cohort_path: Path,
) -> ExdivInputs:
    cohort_bytes = cohort_path.read_bytes()  # before any provider call
    symbols = sorted(
        {s.strip().upper() for s in cohort_symbols(cohort_path) if SUPPORTED_SYMBOL.fullmatch(s.strip().upper())}
    )
    cohort = {"path": str(cohort_path), "sha256": hashlib.sha256(cohort_bytes).hexdigest(), "symbols": len(symbols)}

    first, _last = entry.window.decisions
    bars_through = entry.window.bars_through
    days = await calendar.get_calendar_range(first - _CALENDAR_PAD, bars_through)
    trading_days = tuple(sorted(day.date for day in days if day.is_trading_day))

    actions, actions_manifest = await fetch_corporate_actions_cached(
        actions_source, symbols, entry.actions_window[0], entry.actions_window[1], cache_dir, pace=pace
    )

    bar_dir, raw_dir = cache_dir / "bars", cache_dir / "bars_raw"
    start = datetime.combine(first - _CALENDAR_PAD, time.min, tzinfo=UTC)
    end = datetime.combine(bars_through, time.max, tzinfo=UTC)
    _claim_cache_range(bar_dir, start, end)
    _claim_cache_range(raw_dir, start, end)

    daily: dict = {}
    liquidity_daily: dict = {}
    failures: dict[str, str] = {}
    for symbol in symbols:
        frame, failure = await _bars_or_failure(symbol, "1d", bars, bar_dir, start, end, pace, chunk=end - start)
        if frame is None:
            _record(failures, symbol, failure or "daily: unavailable")
        else:
            daily[symbol] = frame
    for symbol in sorted({*symbols, *static_symbols}):
        frame, failure = await _bars_or_failure(
            symbol, "1d", bars, raw_dir, start, end, pace, adjustment="raw", chunk=end - start
        )
        if frame is None:
            _record(failures, symbol, failure or "daily_raw: unavailable")
        else:
            liquidity_daily[symbol] = frame

    market = MarketData(
        trading_days,
        daily,
        tuple(s for s in static_symbols if s in liquidity_daily),
        liquidity_daily=liquidity_daily,
    )
    events, _ = await asyncio.to_thread(build_exdiv_events, actions, market, entry)
    hourly: dict = {}
    for symbol, group in events.groupby("symbol"):
        span_start = min(group["decision_at"]) - _HOURLY_LEAD
        span_end = min(max(group["decision_at"]) + _HOURLY_TAIL, end)
        frame, failure = await _bars_or_failure(str(symbol), "1h", bars, bar_dir, span_start, span_end, pace)
        if frame is None:
            _record(failures, str(symbol), failure or "hourly: unavailable")
        else:
            hourly[str(symbol)] = frame
    return ExdivInputs(
        actions=actions,
        actions_manifest=actions_manifest,
        cohort=cohort,
        market=market,
        hourly=hourly,
        bar_failures=failures,
        static_requested=tuple(static_symbols),
    )
```
